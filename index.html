submitBtnEl.onclick = function(event) { // Запрещаем стандартное поведение ссылки, чтобы браузер никуда не прыгал сам event.preventDefault(); const phone = phoneInput.value.trim(); const address = addressInput.value.trim(); if ((burgerQty + shawarmaQty) === 0) { alert("Пожалуйста, добавьте хотя бы одно блюдо в заказ."); return false; } if (!phone || !address) { alert("Пожалуйста, введите ваш номер телефона и адрес доставки."); return false; } let total = (burgerQty * burgerPrice) + (shawarmaQty * shawarmaPrice); let text = "*Новый заказ из Papa Martin* 📋\n\n"; if (burgerQty > 0) text += `▪️ Бургер Говядина x ${burgerQty} шт. (${burgerQty * burgerPrice} сом)\n`; if (shawarmaQty > 0) text += `▪️ Шаурма Куриная x ${shawarmaQty} шт. (${shawarmaQty * shawarmaPrice} сом)\n`; text += `\n*Итоговая сумма:* ${total} сом\n`; if (total > 1500) { text += `🚚 *Доставка:* Бесплатная\n\n`; } else { text += `🚚 *Доставка:* По тарифу\n\n`; } text += `👤 *Контакты:* ${phone}\n`; text += `📍 *Адрес доставки:* ${address}`; // Формируем чистую ссылку для WhatsApp let whatsappUrl = "https://whatsapp.com" + encodeURIComponent(text); // Перенаправляем пользователя напрямую через оконный метод — это обходит любые блокировки «about:blank» window.location.href = whatsappUrl; };
